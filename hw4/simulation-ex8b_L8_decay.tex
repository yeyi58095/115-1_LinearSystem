\textbf{MATLAB code:}
\begin{lstlisting}
p = [2 1 1 1 0.5 1];
model = cart_pendulum_model(p, 2);
[V,D] = eig(model.A); lambda = diag(D);
idx=find(-1*real(lambda)>1e-8 & abs(imag(lambda))<1e-8);
[~,j]=min(abs(real(lambda(idx)))); j=idx(j);
v=real(V(:,j)); [~,k]=max(abs(v(2:3))); k=k+1;
if v(k)<0, v=-v; end
z0=scale_angles(v,1);
r=simulate_case(model,z0,10,'ex8b_L8_decay',options,real(lambda(j)));
export_case(r,outputDir,rootDir);
\end{lstlisting}
