\textbf{MATLAB code:}
\begin{lstlisting}
p = [2 1 1 1 0.5 1];
model = cart_pendulum_model(p, 1);
[V,D] = eig(model.A); lambda = diag(D);
idx = find(imag(lambda)>1e-8);
[~,j] = min(imag(lambda(idx))); j=idx(j);
v=V(:,j); [~,k]=max(abs(v(1:3)));
v=v*exp(-1i*angle(v(k)));
z0=scale_angles(real(v),1);
T=2*pi/imag(lambda(j));
r=simulate_case(model,z0,4*T,'ex8a_L7_periodic',options);
export_case(r,outputDir,rootDir);
\end{lstlisting}
